    if (root == NULL)
        return 0;
    queue<HackNode *> HackQueue;
    HackQueue.push(root);
    int Hresult = 0;

        int count = HackQueue.size();
        Hresult = max(Hresult, count);
        for (int i = 0; i < count; i++) {
            HackNode *curr = HackQueue.front();
            HackQueue.pop();
            if (curr->left != NULL)
                HackQueue.push(curr->left);
            if (curr->right != NULL)
                HackQueue.push(curr->right);
        }
    }
    return Hresult;
}
